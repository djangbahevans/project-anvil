add_subdirectory(core)
add_subdirectory(platform)
add_subdirectory(exthost)

if(ANVIL_BUILD_UI STREQUAL "OFF")
  set(ANVIL_UI_ENABLED OFF)
else()
  if(ANVIL_BUILD_UI STREQUAL "ON")
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
  else()
    find_package(Qt6 6.4 QUIET COMPONENTS Widgets)
  endif()
  if(Qt6_FOUND)
    set(ANVIL_UI_ENABLED ON)
  else()
    set(ANVIL_UI_ENABLED OFF)
    message(STATUS "Qt 6.4+ Widgets not found; skipping the UI and application targets")
  endif()
endif()

if(ANVIL_UI_ENABLED)
  add_subdirectory(ui)
  add_subdirectory(app)
endif()
